184 void load_labels_from_yaml(const std::string& file_path)

186 std::ifstream f(file_path.c_str());

191
192 try {
193 YAML::Node config = YAML::LoadFile(file_path);
194
195 auto load_rules = [&](const YAML::Node& node, std::vector<LabelRule>& target) {
196 for (const auto& label_node : node) {
197 std::string label = label_node.as<std::string>();
198 bool is_prefix = false;
199
200 // Se termina com '_', marcar como prefixo
201 if (!label.empty() && label.back() == '_') {
202 is_prefix = true;
203 }
204
205 target.push_back({label, is_prefix});
206 }
207 };
208
209 if (config["authorized_labels"]) {
210 load_rules(config["authorized_labels"], authorized_labels_);

212 }

215 load_rules(config["unauthorized_labels"], unauthorized_labels_);
216 RCLCPP_INFO(this->get_logger(), "%zu labels não autorizados carregados.", unauthorized_labels_.size());
217 }

349 void detectionCallback(const vision_msgs::msg::Detection3DArray::SharedPtr msg)
350 {
351 if (msg->detections.empty())
352 {
353 RCLCPP_WARN(this->get_logger(), "Detection3DArray vazio recebido.");
354 return;
355 }
356
357 for (size_t i = 0; i < msg->detections.size(); ++i)
358 {
359 const auto &det = msg->detections[i];
360 std::string object_id = det.results[0].hypothesis.class_id;
361
362 if (!is_authorized(object_id)) continue;
363
364 geometry_msgs::msg::Pose pose = det.bbox.center;
365 pose.position.z += det.bbox.size.z / 2;
366
367 std::array<double, 3> size_array = {
368 det.bbox.size.x,
369 det.bbox.size.y,
370 det.bbox.size.z
371 };
372
373 if(added.find(object_id) == added.end())
374 {
375 add_collision_box(object_id, size_array, pose);
376 added.insert(object_id);
377 }
378 else
379 {
380 move_collision_box(object_id, pose);
381 }
382 }
