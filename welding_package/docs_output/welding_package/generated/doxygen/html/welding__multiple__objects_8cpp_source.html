221 void loadLocationsFromYaml(const std::string &yaml_path)
222 {
223 try
224 {
225 YAML::Node config = YAML::LoadFile(yaml_path);
226
227 // Agora 'config' é um MAPA: { "trashcan": [...], "firecabinet": [...] }
228 for (const auto &label_node : config)
229 {
230 const std::string label = label_node.first.as<std::string>();
231 const YAML::Node &locations_node = label_node.second;
232
233 std::vector<geometry_msgs::msg::Pose> locations;
234
235 // locations_node é uma LISTA
236 for (const auto &loc_item : locations_node)
237 {
238 if (!loc_item.IsMap() || loc_item.size() != 1)
239 {
240 RCLCPP_WARN(rclcpp::get_logger("yaml_loader"),
241 "[%s] Ignorando entrada inválida de localização.", label.c_str());
242 continue;
243 }
244
245 // Cada item é um mapa { "locationX": { position, orientation } }
246 const auto &loc_name = loc_item.begin()->first.as<std::string>();
247 const YAML::Node &loc_data = loc_item.begin()->second;
248
249 if (!loc_data["position"] || !loc_data["orientation"])
250 {
251 RCLCPP_WARN(rclcpp::get_logger("yaml_loader"),
252 "[%s] '%s' missing position/orientation",
253 label.c_str(), loc_name.c_str());
254 continue;
255 }
256
257 const YAML::Node &pos = loc_data["position"];
258 const YAML::Node &ori = loc_data["orientation"];
259
260 if (pos.size() != 3 || ori.size() != 4)
261 {
262 RCLCPP_WARN(rclcpp::get_logger("yaml_loader"),
263 "[%s] '%s' invalid position/orientation size",
264 label.c_str(), loc_name.c_str());
265 continue;
266 }
267
268 geometry_msgs::msg::Pose pose;
269 pose.position.x = pos[0].as<double>();
270 pose.position.y = pos[1].as<double>();
271 pose.position.z = pos[2].as<double>();
272 pose.orientation.x = ori[0].as<double>();
273 pose.orientation.y = ori[1].as<double>();
274 pose.orientation.z = ori[2].as<double>();
275 pose.orientation.w = ori[3].as<double>();
276
277 locations.push_back(pose);
278
279 RCLCPP_INFO(rclcpp::get_logger("yaml_loader"),
280 "Loaded [%s - %s] -> pos:[%.2f, %.2f, %.2f], ori:[%.2f, %.2f, %.2f, %.2f]",
281 label.c_str(), loc_name.c_str(),
282 pose.position.x, pose.position.y, pose.position.z,
283 pose.orientation.x, pose.orientation.y,
284 pose.orientation.z, pose.orientation.w);
285 }
286
287 welding_poses[label] = locations;
288 }
289 }
290 catch (const YAML::Exception &e)
291 {
292 RCLCPP_ERROR(rclcpp::get_logger("yaml_loader"),
293 "Failed to load YAML file '%s': %s", yaml_path.c_str(), e.what());
294 }

587 void detectionCallback(const vision_msgs::msg::Detection3DArray::SharedPtr msg)
588 {
589 std::string id;
590 for (const auto &det : msg->detections)
591 {
592
593
594 size_t pos = det.results[0].hypothesis.class_id.find('_');
595 if (pos != std::string::npos)
596 {
597 id = det.results[0].hypothesis.class_id.substr(0, pos);
598 }
599 else
600 {
601 id = det.results[0].hypothesis.class_id;
602 }
603
604 if (det.results.empty() || welding_poses.find(id) == welding_poses.end())
605 {
606 continue;
607 }
608
609 if(stopped == false)
610 {
611 publish_velocity(0.2);
612 publish_angular_velocity(0.4);
613 }
614
615
616 if(det.bbox.center.position.y < 0.2 && det.bbox.center.position.y > -0.1 && det.bbox.center.position.x > 0.0 && stopped == true && welding_id == det.results[0].hypothesis.class_id)
617 {
618
619 if (welding_poses.find(id) != welding_poses.end())
620 {
621 const auto &poses = welding_poses[id];
622
623 for (size_t i = 0; i < poses.size(); ++i)
624 {
625 const auto &pose_local = poses[i];
626
627 tf2::Vector3 local_corner(
628 pose_local.position.x,
629 pose_local.position.y,
630 pose_local.position.z);
631
632 const auto &bbox_pose = det.bbox.center;
633
634 tf2::Quaternion q(
635 bbox_pose.orientation.x,
636 bbox_pose.orientation.y,
637 bbox_pose.orientation.z,
638 bbox_pose.orientation.w);
639
640 tf2::Matrix3x3 rot(q);
641 tf2::Vector3 translation(
642 bbox_pose.position.x,
643 bbox_pose.position.y,
644 bbox_pose.position.z);
645
646 tf2::Vector3 world_corner = rot * local_corner + translation;
647
648 geometry_msgs::msg::Pose target_pose;
649 target_pose.position.x = world_corner.x();
650 target_pose.position.y = world_corner.y();
651 target_pose.position.z = world_corner.z();
652
653 target_pose.orientation = pose_local.orientation;
654
655 RCLCPP_INFO(this->get_logger(),
656 "Pose %zu - ponto global: x=%.3f, y=%.3f, z=%.3f",
657 i, world_corner.x(), world_corner.y(), world_corner.z());
658
659 positions_for_arm(target_pose);
660 }
661 }
662 else
663 {
664 RCLCPP_WARN(this->get_logger(), "ID '%s' não encontrado em welding_poses", det.results[0].hypothesis.class_id.c_str());
665 }
666
667 welding_done = true;
668 return_to_welding_position();
669 stopped = false;
670
671 publish_velocity(0.2);
672 publish_angular_velocity(0.4);
673 rclcpp::sleep_for(std::chrono::milliseconds(50));
674 }
675 else if(det.bbox.center.position.y < 0.2 && det.bbox.center.position.y > -0.1 && det.bbox.center.position.x > 0.0 && stopped == false && welding_id != det.results[0].hypothesis.class_id)
676 {
677 publish_velocity(0.0);
678 publish_angular_velocity(0.0);
679 rclcpp::sleep_for(std::chrono::milliseconds(1000));
680 welding_id = det.results[0].hypothesis.class_id;
681 welding_done = false;
682 stopped = true;
683 }
684
685 }
