444 void detectionCallback(const vision_msgs::msg::Detection3DArray::SharedPtr msg)
445 {
446 std::string id;
447 for (const auto &det : msg->detections)
448 {
449 if (det.results.empty())
450 continue;
451
452 size_t pos = det.results[0].hypothesis.class_id.find('_');
453 if (pos != std::string::npos)
454 id = det.results[0].hypothesis.class_id.substr(0, pos);
455 else
456 id = det.results[0].hypothesis.class_id;
457
458 if (welding_poses.find(id) == welding_poses.end())
459 continue;
460
461 if (!stopped)
462 {
463 publish_velocity(0.2);
464 publish_angular_velocity(0.4);
465 }
466
467 if (det.bbox.center.position.y < 0.3 && det.bbox.center.position.y > -0.1 &&det.bbox.center.position.x > 0.0 && stopped && welding_id == det.results[0].hypothesis.class_id)
468 {
469 const auto &poses = welding_poses[id];
470
471 for (const auto &wp_data : poses)
472 {
473 const auto &pose_local = wp_data.pose;
474 const auto &traj_type = wp_data.trajectory_type;
475 const auto &traj_data = wp_data.trajectory_data;
476
477 double raio = 0.1;
478 double angulo_inicial = 0.0;
479 double angulo_final = 2 * M_PI;
480 int num_pontos = 40;
481
482 if (traj_type == "circle" && traj_data.size() == 4)
483 {
484 raio = traj_data[0];
485 angulo_inicial = traj_data[1];
486 angulo_final = traj_data[2];
487 num_pontos = static_cast<int>(traj_data[3]);
488 }
489
490 const auto &bbox_pose = det.bbox.center;
491
492 tf2::Quaternion q_bbox(
493 bbox_pose.orientation.x,
494 bbox_pose.orientation.y,
495 bbox_pose.orientation.z,
496 bbox_pose.orientation.w);
497
498 tf2::Matrix3x3 rot_bbox(q_bbox);
499 tf2::Vector3 translation_bbox(
500 bbox_pose.position.x,
501 bbox_pose.position.y,
502 bbox_pose.position.z);
503
504 tf2::Vector3 local_center(
505 pose_local.position.x,
506 pose_local.position.y,
507 pose_local.position.z);
508
509 tf2::Vector3 centro_mundo = rot_bbox * local_center + translation_bbox;
510
511 tf2::Quaternion q_local(
512 pose_local.orientation.x,
513 pose_local.orientation.y,
514 pose_local.orientation.z,
515 pose_local.orientation.w);
516
517 tf2::Quaternion q_total = q_bbox * q_local;
518 tf2::Matrix3x3 rot_total(q_total);
519
520 RCLCPP_INFO(this->get_logger(),
521 "[%s] Trajetória tipo '%s' - Centro: [%.3f, %.3f, %.3f]",
522 id.c_str(), traj_type.c_str(),
523 centro_mundo.x(), centro_mundo.y(), centro_mundo.z());
524
525 std::vector<geometry_msgs::msg::Pose> waypoints;
526
527 if (traj_type == "circle")
528 {
529 waypoints.reserve(num_pontos);
530 for (int j = 0; j <= num_pontos; ++j)
531 {
532 double t = static_cast<double>(j) / num_pontos;
533 double ang = angulo_inicial + t * (angulo_final - angulo_inicial);
534 tf2::Vector3 ponto_local(raio * cos(ang), raio * sin(ang), 0.0);
535 tf2::Vector3 ponto_mundo = rot_total * ponto_local + centro_mundo;
536
537 geometry_msgs::msg::Pose p;
538 p.position.x = ponto_mundo.x();
539 p.position.y = ponto_mundo.y();
540 p.position.z = ponto_mundo.z();
541 p.orientation = tf2::toMsg(q_total);
542 waypoints.push_back(p);
543 }
544
545 moveit_msgs::msg::RobotTrajectory trajectory;
546 double fraction = move_group_arm->computeCartesianPath(waypoints, 0.01, trajectory);
547
548 if (fraction > 0.99)
549 {
550 moveit::planning_interface::MoveGroupInterface::Plan plan;
551 plan.trajectory = trajectory;
552 move_group_arm->execute(plan);
553 }
554 }
555 else if (traj_type == "line")
556 {
557 geometry_msgs::msg::Pose start_pose = move_group_arm->getCurrentPose().pose;
558
559 geometry_msgs::msg::Pose target_pose = pose_local;
560
561 std::vector<geometry_msgs::msg::Pose> waypoints = {start_pose, target_pose};
562
563 moveit_msgs::msg::RobotTrajectory trajectory;
564 double fraction = move_group_arm->computeCartesianPath(waypoints, 0.01, trajectory);
565
566 if (fraction > 0.99)
567 {
568 moveit::planning_interface::MoveGroupInterface::Plan plan;
569 plan.trajectory = trajectory;
570 move_group_arm->execute(plan);
571 }
572 else
573 {
574 RCLCPP_WARN(this->get_logger(),
575 "Trajetória linear parcial. Fração: %.2f", fraction);
576 }
577 }
578 else if (traj_type == "normal")
579 {
580 const auto &pose_local = wp_data.pose;
581
582 tf2::Vector3 local_corner(
583 pose_local.position.x,
584 pose_local.position.y,
585 pose_local.position.z);
586
587 const auto &bbox_pose = det.bbox.center;
588
589 tf2::Quaternion q(
590 bbox_pose.orientation.x,
591 bbox_pose.orientation.y,
592 bbox_pose.orientation.z,
593 bbox_pose.orientation.w);
594
595 tf2::Matrix3x3 rot(q);
596 tf2::Vector3 translation(
597 bbox_pose.position.x,
598 bbox_pose.position.y,
599 bbox_pose.position.z);
600
601 tf2::Vector3 world_corner = rot * local_corner + translation;
602
603 geometry_msgs::msg::Pose target_pose;
604 target_pose.position.x = world_corner.x();
605 target_pose.position.y = world_corner.y();
606 target_pose.position.z = world_corner.z();
607
608 target_pose.orientation = pose_local.orientation;
609
610 positions_for_arm(target_pose);
611 }
612 }
613
614
615
616 welding_done = true;
617 rclcpp::sleep_for(std::chrono::milliseconds(200));
618 return_to_welding_position();
619 stopped = false;
620
621 publish_velocity(0.2);
622 publish_angular_velocity(0.4);
623 }
624 else if (det.bbox.center.position.y < 0.3 && det.bbox.center.position.y > -0.1 &&
625 det.bbox.center.position.x > 0.0 && !stopped &&
626 welding_id != det.results[0].hypothesis.class_id)
627 {
628 publish_velocity(0.0);
629 publish_angular_velocity(0.0);
630 rclcpp::sleep_for(std::chrono::milliseconds(1000));
631 welding_id = det.results[0].hypothesis.class_id;
632 welding_done = false;
633 stopped = true;
634 }
635 }
