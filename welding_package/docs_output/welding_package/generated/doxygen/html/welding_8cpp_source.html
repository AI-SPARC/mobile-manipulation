168 std::vector<geometry_msgs::msg::Pose> loadLocationsFromYaml(const std::string &yaml_path)
169 {
170
171
172 try
173 {
174 YAML::Node config = YAML::LoadFile(yaml_path);
175
176 for (const auto &it : config)
177 {
178 const std::string key = it.first.as<std::string>();
179 const YAML::Node node = it.second;
180
181 if (!node["position"] || !node["orientation"])
182 {
183 RCLCPP_WARN(rclcpp::get_logger("yaml_loader"),
184 "Location '%s' missing position or orientation", key.c_str());
185 continue;
186 }
187
188 const YAML::Node pos = node["position"];
189 const YAML::Node ori = node["orientation"];
190
191 if (pos.size() != 3 || ori.size() != 4)
192 {
193 RCLCPP_WARN(rclcpp::get_logger("yaml_loader"),
194 "Invalid size for position/orientation in '%s'", key.c_str());
195 continue;
196 }
197
198 geometry_msgs::msg::Pose pose;
199 pose.position.x = pos[0].as<double>();
200 pose.position.y = pos[1].as<double>();
201 pose.position.z = pos[2].as<double>();
202
203 pose.orientation.x = ori[0].as<double>();
204 pose.orientation.y = ori[1].as<double>();
205 pose.orientation.z = ori[2].as<double>();
206 pose.orientation.w = ori[3].as<double>();
207
208 locations.push_back(pose);
209
210 RCLCPP_INFO(rclcpp::get_logger("yaml_loader"),
211 "Loaded %s -> pos:[%.2f, %.2f, %.2f], ori:[%.2f, %.2f, %.2f, %.2f]",
212 key.c_str(),
213 pose.position.x, pose.position.y, pose.position.z,
214 pose.orientation.x, pose.orientation.y,
215 pose.orientation.z, pose.orientation.w);
216 }
217 }
218 catch (const YAML::Exception &e)
219 {
220 RCLCPP_ERROR(rclcpp::get_logger("yaml_loader"),
221 "Failed to load YAML file '%s': %s", yaml_path.c_str(), e.what());
222 }
223

375 void detectionCallback(const vision_msgs::msg::Detection3DArray::SharedPtr msg)
376 {
377 for (const auto &det : msg->detections)
378 {
379 if (det.results.empty() || det.results[0].hypothesis.class_id != "firecabinet")
380 continue;
381
382 if(det.bbox.center.position.y < 0.2 && det.bbox.center.position.y > 0.0 && stopped == false && welding_done.find(det.results[0].hypothesis.class_id) == welding_done.end())
383 {
384 publish_velocity(0.0);
385 rclcpp::sleep_for(std::chrono::milliseconds(500));
386 stopped = true;
387 }
388 else if(stopped == true && welding_done.find(det.results[0].hypothesis.class_id) == welding_done.end())
389 {
390
391 for (size_t i = 0; i < locations.size(); i++)
392 {
393 tf2::Vector3 local_corner(locations[i].position.x, locations[i].position.y, locations[i].position.z);
394
395 const auto &pose = det.bbox.center;
396 tf2::Quaternion q(pose.orientation.x, pose.orientation.y, pose.orientation.z, pose.orientation.w);
397 tf2::Matrix3x3 rot(q);
398 tf2::Vector3 translation(pose.position.x, pose.position.y, pose.position.z);
399
400 tf2::Vector3 world_corner = rot * local_corner + translation;
401
402 geometry_msgs::msg::Pose target_pose;
403 target_pose.position.x = world_corner.x();
404 target_pose.position.y = world_corner.y();
405 target_pose.position.z = world_corner.z();
406
407
408 target_pose.orientation.x = locations[i].orientation.x;
409 target_pose.orientation.y = locations[i].orientation.y;
410 target_pose.orientation.z = locations[i].orientation.z;
411 target_pose.orientation.w = locations[i].orientation.w;
412
413 RCLCPP_INFO(this->get_logger(),
414 "Pose %zu - ponto global: x=%.3f, y=%.3f, z=%.3f",
415 i, world_corner.x(), world_corner.y(), world_corner.z());
416
417 positions_for_arm(target_pose);
418 }
419
420 welding_done.insert(det.results[0].hypothesis.class_id);
421 return_to_welding_position();
422 stopped = false;
423 publish_velocity(0.1);
424 rclcpp::sleep_for(std::chrono::milliseconds(50));
425 }
426 else
427 {
428 publish_velocity(0.1);
429 rclcpp::sleep_for(std::chrono::milliseconds(50));
430 }
431 }
